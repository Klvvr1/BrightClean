# lib/features/agent/domain/usecases/ — Use Cases (Business Logic)
# =====================================================================
# Use Cases contain PURE business logic. They:
#   - Receive input from the Presentation layer (UI/BLoC/Controller)
#   - Call Repository interfaces (defined in domain/repositories/)
#   - Return results to the Presentation layer
#   - Have NO dependency on Flutter, databases, or APIs
#
# This is the heart of Clean Architecture. A UseCase represents
# a single, well-defined action the application can perform.
#
# --- Dart Pseudo-Code Example: AcceptOrderUseCase ---
#
# import '../repositories/agent_repository.dart';
#
# class AcceptOrderUseCase {
#   final AgentRepository repository;
#
#   const AcceptOrderUseCase({required this.repository});
#
#   /// Returns a result (or throws a domain-level Failure).
#   Future<void> call(/* params */) async {
#     return await repository./* relevant method */();
#   }
# }
